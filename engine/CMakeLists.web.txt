# LeirEngineCore — web-safe static subset of the engine (Fase 6 / M2).
# Renders through the RHI (WebGPUBackend + BackendFactory); excludes the
# Vulkan/D3D12 backends, Jolt physics (PhysicsWorld.web.cpp no-ops the step)
# and SoLoud audio. The desktop engine keeps its own target in CMakeLists.txt.
# This file is include()d by the standalone Emscripten projects
# (examples/WebEngineDemo), which must already have fetched glm, nlohmann_json
# and stb with the same tags as dependencies/CMakeLists.txt.

target_include_directories(LeirEngineCore
    PUBLIC
        ${LEIR_ROOT}/engine/include
        ${glm_SOURCE_DIR}
    PRIVATE
        ${LEIR_ROOT}/engine/src
        ${stb_SOURCE_DIR}
        ${LEIR_ROOT}/dependencies/emdawnwebgpu/webgpu/include
)

target_link_libraries(LeirEngineCore PUBLIC nlohmann_json::nlohmann_json)

target_compile_definitions(LeirEngineCore PUBLIC
    LEIR_SHADER_DIR="/shaders"
    PRIVATE LEIR_ENGINE_BUILD
)

target_compile_options(LeirEngineCore PRIVATE
    -sASYNCIFY=1
    --use-port=contrib.glfw3
)

set_target_properties(LeirEngineCore PROPERTIES CXX_VISIBILITY_PRESET hidden)
